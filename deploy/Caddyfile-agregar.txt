# ─────────────────────────────────────────────────────────────────────────
#  La tienda cuelga de  motoivir.com/caseritos
#
#  REEMPLAZA el bloque de motoivir.com que ya tenés por este.
#  No hace falta tocar el DNS: es el mismo dominio de siempre.
#
#  Despues:  systemctl reload caddy
# ─────────────────────────────────────────────────────────────────────────

motoivir.com, www.motoivir.com {
	encode gzip

	# Sin la barra final, las rutas relativas de la tienda apuntarian al
	# dominio raiz y no cargaria nada. Este redir lo evita.
	redir /caseritos /caseritos/

	# handle_path QUITA el prefijo antes de pasar la peticion:
	# el servidor de la tienda recibe "/productos.json", no
	# "/caseritos/productos.json", y por eso no hay que tocar su codigo.
	handle_path /caseritos/* {
		reverse_proxy localhost:3001
	}

	# Todo lo demas sigue yendo a MOTO-IVIR, igual que siempre.
	handle {
		reverse_proxy localhost:3000
	}
}

# ─────────────────────────────────────────────────────────────────────────
#  El orden importa: handle_path (mas especifico) va ANTES que handle.
#  Caddy evalua los bloques handle en el orden en que aparecen.
# ─────────────────────────────────────────────────────────────────────────
